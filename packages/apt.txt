# System packages, one per line, with the reason it is needed.
# Consumed by `install.sh --deps`; comments and blank lines are ignored.
#
# Keep the reason column filled in. The friction this file exists to remove is
# not installing things, it is not knowing something was required in the first
# place.

# --- core ---
git                 # everything
tmux                # tmux/.tmux.conf
# NOT listed for apt on purpose. Debian, Ubuntu LTS and Raspberry Pi OS all
# ship a Neovim below the 0.11 that AstroNvim v6 requires, and an apt install
# would satisfy `command -v nvim` while still being unusable. install.sh
# fetches an official build into ~/.local instead. See packages/manual.md.
ccache              # ccache/ccache.conf, USE_CCACHE in bash/common.sh
bash-completion     # sourced by bash/bashrc.linux

# --- nvim: snacks.image ---
# ImageMagick 6 is what Ubuntu ships. It provides convert/identify but no
# `magick` binary; snacks.image detects this and falls back, so 6 is enough.
imagemagick         # snacks.image -- without it images never decode
ghostscript         # snacks.image: PDF rendering (optional)

# --- mason (nvim LSP/DAP/tool installer) ---
# Mason downloads prebuilt tools and builds others, and it fails per-package
# with only "failed to install" in the UI -- the real cause is in :MasonLog.
# These are the prerequisites whose absence has actually bitten here.
unzip               # clangd and other zip-packaged tools; failed on this exact cause
python3-venv        # python-based servers (nginx-language-server, ruff, basedpyright).
                    # Mason invokes the SYSTEM python3.12, not whatever python3
                    # resolves to on PATH, so a uv- or pyenv-managed python having
                    # venv is not enough.

# --- terminal ---
fzf                 # tmux prefix+s session switcher, tmux-fzf plugin
entr                # tmux-autoreload plugin -- silently inert without it
ncurses-term        # terminfo entries incl. tmux-256color

# --- notifications (WSL/Linux desktop) ---
# Provides org.freedesktop.Notifications on the session bus. nvim and other
# apps raise notifications over D-Bus; without a daemon those calls hang until
# they time out rather than failing fast.
dunst               # notification daemon
libnotify-bin       # notify-send, for testing and for scripts

# --- browser (headless, for mermaid) ---
# mermaid-cli drives a headless browser through puppeteer. Installing Chrome
# from the distro and pointing puppeteer at it avoids a second ~150MB Chromium
# download; bash/common.sh exports PUPPETEER_EXECUTABLE_PATH.
# Chrome is not in the Ubuntu archive -- see packages/manual.md if absent.

# --- LaTeX (vimtex + texlab) ---
# The practical TeX Live set, about 700 MB with dependencies against 7 GB for
# texlive-full. pdfLaTeX is the engine Overleaf, arXiv and the IEEE/ACM
# templates are built against, which is why documents compile with this rather
# than tectonic; tectonic stays only for snacks.image's inline math. TikZ comes
# in as a dependency (texlive-pictures). If a template needs a .sty this lacks,
# `apt-file search <name>.sty` names the collection that provides it.
latexmk                    # vimtex compiler -- rebuilds on every save
texlive-latex-recommended  # core classes plus booktabs, microtype, hyperref
texlive-latex-extra        # the long tail real templates pull in: cleveref, tcolorbox, geometry
texlive-fonts-recommended  # fonts standard templates assume
texlive-publishers         # journal and conference classes: IEEEtran, acmart
texlive-bibtex-extra       # biblatex and extra bibliography styles
texlive-science            # siunitx, algorithm2e, algorithmicx, pgfplots
biber                      # biblatex backend; bibtex alone cannot drive biblatex
chktex                     # LaTeX linter, surfaced as texlab diagnostics
zathura                    # vimtex PDF viewer, synctex in both directions
zathura-pdf-poppler        # zathura's PDF backend -- it opens nothing without one

# tectonic and mmdc are NOT apt packages. See packages/manual.md.
